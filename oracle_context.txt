"""Dashboard account UI: login/register in the sidebar and a personalised feed tab."""
import streamlit as st

from dashboard import api_client, utils


def _sign_in(action, email: str, password: str) -> None:
    try:
        result = action(email.strip(), password)
    except api_client.ApiError as e:
        st.warning(str(e))
        return
    st.session_state["token"] = result["access_token"]
    st.session_state["user_email"] = result["user"]["email"]
    st.rerun()


def _sign_out() -> None:
    for key in ("token", "user_email"):
        st.session_state.pop(key, None)
    st.rerun()


def render_sidebar() -> None:
    st.subheader("Account")
    if st.session_state.get("token"):
        st.caption(f"Logged in: {st.session_state.get('user_email', '')}")
        if st.button("Logout"):
            _sign_out()
        return
    email = st.text_input("Email", key="acc_email")
    password = st.text_input("Password (kam az kam 8 harf)", type="password", key="acc_password")
    left, right = st.columns(2)
    if left.button("Login"):
        _sign_in(api_client.login, email, password)
    if right.button("Register"):
        _sign_in(api_client.register, email, password)


HOUR_CHOICES = ["Main digest ke saath"] + [f"{h:02d}:00 UTC" for h in range(24)]


def _render_delivery(token: str) -> None:
    try:
        delivery = api_client.get_delivery(token)
    except api_client.ApiError as e:
        st.warning(str(e))
        return
    hour = delivery.get("digest_hour")
    with st.form("delivery_form"):
        wanted = st.checkbox(
            f"Daily digest meri email par bhejein ({delivery['email']})",
            value=delivery["digest_enabled"],
        )
        chat = st.text_input(
            "Telegram chat id (optional)",
            value=delivery.get("telegram_chat_id", ""),
            placeholder="123456789 ya @channelname",
        )
        choice = st.selectbox(
            "Digest ka waqt (UTC, Pakistan = UTC+5)",
            HOUR_CHOICES,
            index=0 if hour is None else hour + 1,
        )
        saved = st.form_submit_button("Save delivery settings")
    st.caption("Telegram ke liye pehle apne bot ko /start bhejein, phir apni chat id yahan likhein.")
    if saved:
        new_hour = None if choice == HOUR_CHOICES[0] else HOUR_CHOICES.index(choice) - 1
        try:
            api_client.save_delivery(token, wanted, chat.strip(), new_hour)
            st.success("Delivery settings save ho gayi.")
        except api_client.ApiError as e:
            st.warning(str(e))


def render_feed(limit: int) -> None:
    token = st.session_state.get("token")
    if not token:
        st.info("Apni personal feed ke liye sidebar se Login ya Register karein.")
        return

    only_mine = st.checkbox(
        "Dusre users ki custom sources ki stories chhupayein", key="feed_only_mine"
    )
    scope_args = {"scope": "mine"} if only_mine else {}
    try:
        feed = api_client.my_feed(token, limit=limit, **scope_args)
    except api_client.ApiError as e:
        st.warning(f"Feed nahi mili: {e}")
        if st.button("Logout karein", key="feed_logout"):
            _sign_out()
        return

    st.caption("Ye feed sirf aapki interests ke hisab se rank hoti hai.")
    text = st.text_input(
        "Meri interests (comma se alag)",
        value=", ".join(feed["interests"]),
        placeholder="cricket, ai, pakistan",
        key="my_interests",
    )
    if st.button("Save my interests"):
        try:
            api_client.save_my_interests(token, text)
            st.success("Save ho gaya. Feed dobara load hogi.")
            st.rerun()
        except api_client.ApiError as e:
            st.warning(str(e))

    _render_delivery(token)

    if not feed["stories"]:
        st.info("Abhi koi story nahi. Pehle digest chalayein.")
        return
    for i, story in enumerate(feed["stories"], 1):
        with st.container(border=True):
            badges = [
                utils.badge_html(story["priority"], utils.priority_color(story["priority"])),
                utils.badge_html(story["credibility"], utils.NEUTRAL_COLOR),
            ] + [utils.badge_html("match: " + k, "#6b46c1") for k in story["matched_interests"]]
            st.markdown(" &nbsp; ".join(badges), unsafe_allow_html=True)
            st.markdown(
                f'<div style="font-size:18px;font-weight:600;margin:6px 0;">'
                f"{i}. {utils.html_safe(story['title'])}</div>",
                unsafe_allow_html=True,
            )
            st.markdown(utils.md_safe(story["summary"]))
            if story.get("claims"):
                st.markdown(utils.claims_markdown(story["claims"]))
            st.caption("Sources: " + utils.sources_markdown(story.get("sources", [])))


def render_sources() -> None:
    token = st.session_state.get("token")
    if not token:
        return
    st.divider()
    st.subheader("Meri sources")
    try:
        data = api_client.my_sources(token)
    except api_client.ApiError as e:
        st.warning(str(e))
        return
    st.caption(
        f"{len(data['sources'])}/{data['limit']} sources. Ye sources sab ke digest ke liye "
        "collect hoti hain aur '(custom)' label ke saath dikhti hain."
    )
    for src in data["sources"]:
        left, right = st.columns([5, 1])
        left.text(f"{src['name']} [{src['kind']}] {src['url']}")
        if right.button("Delete", key=f"del_src_{src['id']}"):
            try:
                api_client.delete_my_source(token, src["id"])
                st.rerun()
            except api_client.ApiError as e:
                st.warning(str(e))
    kind = st.selectbox("Type", ["rss", "youtube"], key="src_kind")
    name = st.text_input("Naam", key="src_name")
    url = st.text_input("RSS feed URL ya YouTube channel ID (UC...)", key="src_url")
    if st.button("Source add karein"):
        try:
            api_client.add_my_source(token, kind, url, name)
            st.success("Source add ho gayi.")
            st.rerun()
        except api_client.ApiError as e:
            st.warning(str(e))

import os

import httpx

from app.core.config import settings

DEFAULT_URL = "http://127.0.0.1:8000"
_transport = None  # tests yahan fake transport lagate hain


class ApiError(Exception):
    """API se rabta ya jawab me masla (dashboard ye message seedha dikhata hai)."""


def base_url() -> str:
    return os.getenv("API_URL", DEFAULT_URL).rstrip("/")


def _detail(response: httpx.Response) -> str:
    try:
        detail = response.json().get("detail")
    except (ValueError, AttributeError):
        detail = None
    if isinstance(detail, str) and detail:
        return detail
    return f"API error {response.status_code}"


def _forward_headers() -> dict:
    """Dashboard ke visitor ka IP X-Forwarded-For me forward karti hai (rate limit sahi chale)."""
    try:
        import ipaddress

        import streamlit as st

        raw = st.context.headers.get("x-forwarded-for", "")
        first = raw.split(",")[0].strip()
        ipaddress.ip_address(first)  # galat/ganda value ho to ValueError, hum chhor dete hain
    except Exception:
        return {}
    return {"X-Forwarded-For": first}


def _request(
    method: str, path: str, timeout: float = 20.0, extra_headers: dict | None = None, **kwargs
):
    try:
        headers = {"X-API-Key": settings.api_key} if settings.api_key else {}
        headers.update(_forward_headers())
        headers.update(extra_headers or {})
        with httpx.Client(
            base_url=base_url(), timeout=timeout, transport=_transport, headers=headers
        ) as client:
            response = client.request(method, path, **kwargs)
    except httpx.ConnectError:
        raise ApiError(
            "API server se connection nahi ho paya. Pehle server chalayein: "
            "uvicorn app.main:app --reload"
        ) from None
    except httpx.TimeoutException:
        raise ApiError("API ne waqt par jawab nahi diya (timeout). Dobara try karein.") from None
    except httpx.HTTPError as e:
        raise ApiError(f"Network error: {type(e).__name__}") from None

    if response.status_code >= 400:
        raise ApiError(_detail(response))
    return response.json()


def health() -> dict:
    return _request("GET", "/health", timeout=5)


def get_stories(limit: int = 20, priority: str | None = None, hours: int | None = None) -> dict:
    params: dict = {"limit": limit}
    if priority:
        params["priority"] = priority
    if hours:
        params["hours"] = hours
    return _request("GET", "/stories", params=params)


def get_stats() -> dict:
    return _request("GET", "/stats")


def get_articles(limit: int = 50) -> dict:
    return _request("GET", "/articles", params={"limit": limit})


def ask(question: str) -> dict:
    # Chat me 2-3 Gemini calls lagti hain (pehli baar embeddings bhi), is liye lamba timeout
    return _request("POST", "/chat", timeout=90, json={"question": question})


def run_digest() -> dict:
    return _request("POST", "/run-digest")


def run_alerts() -> dict:
    return _request("POST", "/run-alerts")


def get_videos(limit: int = 10) -> dict:
    return _request("GET", "/videos", params={"limit": limit})


def run_videos() -> dict:
    return _request("POST", "/run-videos")


def get_interests() -> dict:
    return _request("GET", "/interests")


def save_interests(interests: str) -> dict:
    return _request("PUT", "/interests", json={"interests": interests})


def research(topic: str) -> dict:
    # 2 Gemini calls + embeddings, is liye lamba timeout
    return _request("POST", "/research", timeout=120, json={"topic": topic})


def _bearer(token: str) -> dict:
    return {"Authorization": f"Bearer {token}"}


def register(email: str, password: str) -> dict:
    return _request("POST", "/auth/register", json={"email": email, "password": password})


def login(email: str, password: str) -> dict:
    return _request("POST", "/auth/login", json={"email": email, "password": password})


def my_feed(token: str, limit: int = 20, scope: str = "all") -> dict:
    params: dict = {"limit": limit}
    if scope != "all":
        params["scope"] = scope
    return _request("GET", "/me/feed", params=params, extra_headers=_bearer(token))


def save_my_interests(token: str, interests: str) -> dict:
    return _request(
        "PUT", "/me/interests", json={"interests": interests}, extra_headers=_bearer(token)
    )


def get_delivery(token: str) -> dict:
    return _request("GET", "/me/delivery", extra_headers=_bearer(token))


_UNSET = object()


def save_delivery(
    token: str, digest_enabled=_UNSET, telegram_chat_id=_UNSET, digest_hour=_UNSET
) -> dict:
    """Sirf wahi fields bhejti hai jo di gayi hon (digest_hour=None matlab main digest ke saath)."""
    body = {}
    for key, value in (
        ("digest_enabled", digest_enabled),
        ("telegram_chat_id", telegram_chat_id),
        ("digest_hour", digest_hour),
    ):
        if value is not _UNSET:
            body[key] = value
    return _request("PUT", "/me/delivery", json=body, extra_headers=_bearer(token))


def my_sources(token: str) -> dict:
    return _request("GET", "/me/sources", extra_headers=_bearer(token))


def add_my_source(token: str, kind: str, url: str, name: str) -> dict:
    return _request(
        "POST",
        "/me/sources",
        json={"kind": kind, "url": url, "name": name},
        extra_headers=_bearer(token),
    )


def delete_my_source(token: str, source_id: int) -> dict:
    return _request("DELETE", f"/me/sources/{source_id}", extra_headers=_bearer(token))


def web_check(text: str) -> dict:
    # Web search + Gemini, is liye lamba timeout
    return _request("POST", "/research/web", timeout=90, json={"text": text})

def require_password():
    """DASHBOARD_PASSWORD set ho to bina password ke dashboard nahi khulta."""
    password = settings.dashboard_password
    if not password or st.session_state.get("authed"):
        return
    holder = st.empty()
    entered = holder.text_input("Password", type="password")
    if entered and hmac.compare_digest(entered.encode(), password.encode()):
        st.session_state.authed = True
        holder.empty()
        return
    if entered:
        st.error("Ghalat password")
    st.stop()

from datetime import datetime
from html import escape

PRIORITY_COLORS = {
    "Critical": "#c53030",
    "High": "#dd6b20",
    "Medium": "#2b6cb0",
    "Low": "#718096",
}
SENTIMENT_COLORS = {"positive": "#2f855a", "negative": "#c53030", "neutral": "#718096"}
NEUTRAL_COLOR = "#4a5568"


def priority_color(priority: str) -> str:
    return PRIORITY_COLORS.get(priority, PRIORITY_COLORS["Medium"])


def sentiment_color(sentiment: str) -> str:
    return SENTIMENT_COLORS.get(sentiment, SENTIMENT_COLORS["neutral"])


def format_time(iso: str | None) -> str:
    """DB ka waqt UTC me hota hai."""
    if not iso:
        return "-"
    try:
        return datetime.fromisoformat(iso).strftime("%d %b %Y, %H:%M UTC")
    except ValueError:
        return str(iso)


def badge_html(text: str, color: str) -> str:
    return (
        f'<span style="background:{color};color:#fff;padding:2px 10px;'
        f'border-radius:10px;font-size:12px;">{escape(text)}</span>'
    )


def md_safe(text: str) -> str:
    """Markdown me '$' se math ban jata hai (jaise '$1M'), is liye escape."""
    return text.replace("$", "\\$")


def html_safe(text: str) -> str:
    return escape(text).replace("$", "&#36;")


def sources_markdown(sources: list[dict]) -> str:
    parts = []
    for s in sources:
        name = str(s.get("source", "source")).replace("[", "").replace("]", "")
        link = str(s.get("link", ""))
        if link.startswith(("http://", "https://")):
            link = link.replace("(", "%28").replace(")", "%29")
            parts.append(f"[{name}]({link})")
        else:
            parts.append(name)
    return " | ".join(parts)


CLAIM_LABELS = {"confirmed": "confirmed", "disputed": "disputed", "single": "single source"}


def _plain(text) -> str:
    return md_safe(str(text).replace("[", "(").replace("]", ")"))


def claims_markdown(claims) -> str:
    """Story ke claims ki bullet list (confirmed / disputed / single source)."""
    lines = []
    for c in claims or []:
        label = CLAIM_LABELS.get(c.get("status"), "single source")
        who = ", ".join(c.get("supported_by", []))
        against = ", ".join(c.get("contradicted_by", []))
        detail = who + ("; against: " + against if against else "")
        lines.append(f"- {_plain(c.get('claim', ''))} _[{label}: {_plain(detail)}]_")
    return "\n".join(lines)


def link_markdown(text: str, url: str) -> str:
    """Markdown link. Title ke [ ] badal deta hai taake link na toote."""
    label = md_safe(str(text).replace("[", "(").replace("]", ")"))
    if not str(url).startswith(("http://", "https://")):
        return label
    safe = str(url).replace("(", "%28").replace(")", "%29")
    return f"[{label}]({safe})"

def _secret() -> str:
    if not settings.auth_secret:
        raise HTTPException(status_code=503, detail="Accounts band hain: AUTH_SECRET set nahi hai")
    return settings.auth_secret


def _throttle(request: Request, email: str | None = None) -> None:
    """Bruteforce rokne ke liye: IP aur (login par) email, dono par limit."""
    limit, window = settings.auth_rate_limit, settings.auth_rate_window_seconds
    keys = [f"auth-ip:{client_ip(request, settings.trust_proxy_headers)}"]
    if email:
        keys.append(f"auth-email:{email[:254]}")
    for key in keys:
        wait = ratelimit.hit(key, limit, window)
        if wait:
            raise HTTPException(
                status_code=429,
                detail="Bohat zyada koshishein. Thori der baad dobara try karein",
                headers={"Retry-After": str(int(wait) + 1)},
            )


def _issue(user: dict) -> dict:
    token = create_token(user["id"], _secret(), settings.auth_token_hours * 3600)
    return {"access_token": token, "token_type": "bearer", "user": user}


def get_current_user(authorization: str | None = Header(default=None)) -> dict:
    secret = _secret()
    scheme, _, token = (authorization or "").partition(" ")
    user_id = verify_token(token.strip(), secret) if scheme.lower() == "bearer" else None
    user = users.get_user(user_id) if user_id is not None else None
    if user is None:
        raise HTTPException(status_code=401, detail="Login chahiye ya token expire ho gaya")
    return user

            )
        return await call_next(request)

"""Request id, access log aur basic security headers."""
import logging
import re
import time
import uuid

from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse

from app.core import error_alerts
from app.core.logging_setup import request_id_var

log = logging.getLogger("app.http")
_ID_RE = re.compile(r"^[A-Za-z0-9._-]{8,64}$")
SECURITY_HEADERS = {
    "X-Content-Type-Options": "nosniff",
    "X-Frame-Options": "DENY",
    "Referrer-Policy": "no-referrer",
}


def add_http_middleware(app: FastAPI) -> None:
    @app.middleware("http")
    async def request_wrapper(request: Request, call_next):
        incoming = request.headers.get("x-request-id", "")
        request_id = incoming if _ID_RE.match(incoming) else uuid.uuid4().hex
        token = request_id_var.set(request_id)
        start = time.perf_counter()
        fields = {"method": request.method, "path": request.url.path}
        try:
            try:
                response = await call_next(request)
            except Exception as exc:
                log.exception("unhandled error", extra={"fields": fields})
                error_alerts.report(
                    f"Server error 500: {request.method} {request.url.path}",
                    f"request_id={request_id} error={type(exc).__name__}",
                )
                response = JSONResponse(
                    {"detail": "Server me masla aaya", "request_id": request_id}, status_code=500
                )
            fields["status"] = response.status_code
            fields["duration_ms"] = round((time.perf_counter() - start) * 1000, 1)
            level = logging.DEBUG if request.url.path == "/health" else logging.INFO
            log.log(level, "request", extra={"fields": fields})
            response.headers["X-Request-ID"] = request_id
            for name, value in SECURITY_HEADERS.items():
                response.headers.setdefault(name, value)
            return response
        finally:
            request_id_var.reset(token)

def require_api_key(x_api_key: str | None = Header(default=None)) -> None:
    """API_KEY set ho to har request me X-API-Key header chahiye. Khali ho to koi rok nahi."""
    expected = settings.api_key
    if not expected:
        return
    if not x_api_key or not hmac.compare_digest(x_api_key.encode(), expected.encode()):
        raise HTTPException(status_code=401, detail="API key missing ya ghalat hai")

import socket
import time
from dataclasses import dataclass, field
from datetime import datetime
from typing import Optional

import feedparser
from bs4 import BeautifulSoup

from app.core.user_sources import load_extra_feeds

socket.setdefaulttimeout(10)

# Kuch sites (jaise Al Jazeera) feedparser ke default User-Agent ko block karti hain
feedparser.USER_AGENT = (
    "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 "
    "(KHTML, like Gecko) Chrome/124.0 Safari/537.36"
)
RETRY_DELAY = 1.5

FEEDS = {
    "BBC World": "https://feeds.bbci.co.uk/news/world/rss.xml",
    "Dawn": "https://www.dawn.com/feeds/home",
    "The Guardian": "https://www.theguardian.com/world/rss",
    "TechCrunch": "https://techcrunch.com/feed/",
}


@dataclass
class Article:
    title: str
    link: str
    summary: str
    source: str
    published: Optional[datetime] = None
    extra: dict = field(default_factory=dict)


def clean_html(text: str) -> str:
    """HTML tags hata kar plain text return karta hai."""
    if not text:
        return ""
    return BeautifulSoup(text, "html.parser").get_text(" ", strip=True)


def parse_date(entry) -> Optional[datetime]:
    parsed = entry.get("published_parsed") or entry.get("updated_parsed")
    if parsed:
        return datetime(*parsed[:6])
    return None


def fetch_feed(source: str, url: str, limit: int = 10) -> list[Article]:
    """Ek feed se articles laata hai. Fail ho to khali list return karta hai."""
    feed = None
    for attempt in range(2):  # ek baar retry: temporary network hiccup se bachne ke liye
        try:
            feed = feedparser.parse(url)
        except Exception as e:
            print(f"[ERROR] {source}: {e}")
            feed = None
        if feed is not None and feed.entries:
            break
        if attempt == 0:
            time.sleep(RETRY_DELAY)

    if feed is None or not feed.entries:
        print(f"[WARN] {source}: koi article nahi mila")
        return []

    articles = []
    for entry in feed.entries[:limit]:
        articles.append(
            Article(
                title=clean_html(entry.get("title", "")),
                link=entry.get("link", ""),
                summary=clean_html(entry.get("summary", "")),
                source=source,
                published=parse_date(entry),
            )
        )
    return articles


def collect_all(limit_per_feed: int = 10, extra_feeds: dict | None = None) -> list[Article]:
    """Saari feeds se articles collect karta hai."""
    feeds = dict(FEEDS)
    feeds.update(load_extra_feeds() if extra_feeds is None else extra_feeds)
    all_articles: list[Article] = []
    for source, url in feeds.items():
        all_articles.extend(fetch_feed(source, url, limit_per_feed))
    return all_articles


if __name__ == "__main__":
    articles = collect_all(limit_per_feed=5)
    print(f"\nTotal articles: {len(articles)}\n")
    for a in articles:
        print(f"[{a.source}] {a.title}")
        print(f"   {a.link}")
        print(f"   {a.published}\n")
