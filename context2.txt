def save_new_articles(articles: list[Article]) -> list[Article]:
    """DB me sirf naye articles save karta hai, aur wahi return karta hai."""
    init_db()
    new_articles: list[Article] = []

    with SessionLocal() as session:
        existing_links = {row[0] for row in session.query(ArticleDB.link).all()}

        for a in articles:
            if a.link in existing_links:
                continue
            session.add(
                ArticleDB(
                    title=a.title[:500],
                    link=a.link,
                    summary=a.summary,
                    source=a.source[:100],
                    image_url=(getattr(a, "extra", None) or {}).get("image", "")[:1000],
                    published=a.published,
                )
            )
            existing_links.add(a.link)
            new_articles.append(a)

        session.commit()

    return new_articles


def count_articles() -> int:
    init_db()
    with SessionLocal() as session:
        return session.query(ArticleDB).count()


def get_undigested_articles(hours: int = 36) -> list[Article]:
    """Pichle `hours` ghanton ke wo articles jo abhi tak kisi digest me nahi gaye."""
    init_db()
    cutoff = datetime.utcnow() - timedelta(hours=hours)
    with SessionLocal() as session:
        done = {row[0] for row in session.query(DigestedDB.link).all()}
        rows = (
            session.query(ArticleDB)
            .filter(ArticleDB.collected_at >= cutoff)
            .order_by(ArticleDB.collected_at.desc())
            .all()
        )
        return [
            Article(
                title=r.title,
                link=r.link,
                summary=r.summary or "",
                source=r.source,
                published=r.published,
                extra={"image": r.image_url} if r.image_url else {},
            )
            for r in rows
            if r.link not in done
        ]


def mark_digested(articles: list[Article]) -> None:
    """Articles ko 'digest me chala gaya' likh deta hai."""
    if not articles:
        return
    init_db()
    with SessionLocal() as session:
        existing = {row[0] for row in session.query(DigestedDB.link).all()}
        for a in articles:
            link = a.link[:1000]
            if link in existing:
                continue
            session.add(DigestedDB(link=link))
            existing.add(link)
        session.commit()


def story_key(story: Story) -> str:
    """Story ki pehchan: us ka pehla source link."""
    key = story.sources[0][1] if story.sources else story.title
    return key[:1000]


def filter_unalerted(stories: list[Story]) -> list[Story]:
    """Sirf wo stories jin ka alert pehle nahi gaya."""
    if not stories:
        return []
    init_db()
    with SessionLocal() as session:
        done = {row[0] for row in session.query(AlertDB.story_key).all()}
    return [s for s in stories if story_key(s) not in done]


def mark_alerted(stories: list[Story]) -> None:
    """Stories ko 'alert bhej diya' likh deta hai."""
    if not stories:
        return
    init_db()
    with SessionLocal() as session:
        existing = {row[0] for row in session.query(AlertDB.story_key).all()}
        for s in stories:
            key = story_key(s)
            if key in existing:
                continue
            session.add(AlertDB(story_key=key, title=s.title[:500]))
            existing.add(key)
        session.commit()


def save_stories(stories: list[Story]) -> None:
    """Digest ki stories DB me save karta hai (API aur dashboard ke liye)."""
    if not stories:
        return
    init_db()
    now = datetime.utcnow()  # ek digest ki saari stories ka ek hi waqt
    with SessionLocal() as session:
        for s in stories:  # rank ke order me save hoti hain, is liye id se order milta hai
            session.add(
                StoryDB(
                    title=s.title[:500],
                    summary=s.summary,
                    priority=s.priority,
                    sentiment=s.sentiment,
                    credibility=s.credibility,
                    claims_json=json.dumps(getattr(s, "claims", None) or []),
                    image_url=(getattr(s, "image_url", "") or "")[:1000],
                    sources_json=json.dumps([list(x) for x in s.sources]),
                    created_at=now,
                )
            )
        session.commit()


def _story_to_dict(row: StoryDB) -> dict:
    try:
        sources = json.loads(row.sources_json or "[]")
    except ValueError:
        sources = []
    try:
        claims = json.loads(getattr(row, "claims_json", None) or "[]")
    except ValueError:
        claims = []
    if not isinstance(claims, list):
        claims = []
    return {
        "id": row.id,
        "title": row.title,
        "summary": row.summary,
        "priority": row.priority,
        "sentiment": row.sentiment,
        "credibility": row.credibility,
        "claims": claims,
        "image_url": getattr(row, "image_url", "") or "",
        "category": categorize(row.title, row.summary or ""),
        "sources": [{"source": s[0], "link": s[1]} for s in sources if len(s) == 2],
        "created_at": row.created_at.isoformat() if row.created_at else None,
    }


def get_recent_stories(
    limit: int = 20, priority: str | None = None, hours: int | None = None
) -> list[dict]:
    """Sab se naya digest pehle, aur digest ke andar rank ke order me."""
    init_db()
    with SessionLocal() as session:
        query = session.query(StoryDB)
        if priority:
            query = query.filter(StoryDB.priority == priority)
        if hours:
            query = query.filter(StoryDB.created_at >= datetime.utcnow() - timedelta(hours=hours))
        rows = query.order_by(StoryDB.created_at.desc(), StoryDB.id.asc()).limit(limit).all()
        return [_story_to_dict(r) for r in rows]

def get_stats() -> dict:
    init_db()
    with SessionLocal() as session:
        by_priority = {p: 0 for p in PRIORITIES}
        for priority, n in (
            session.query(StoryDB.priority, func.count(StoryDB.id)).group_by(StoryDB.priority).all()
        ):
            by_priority[priority] = n
        return {
            "articles": session.query(ArticleDB).count(),
            "stories": session.query(StoryDB).count(),
            "alerts_sent": session.query(AlertDB).count(),
            "stories_by_priority": by_priority,
        }

def normalize_database_url(url: str) -> str:
    """Render 'postgres://' deta hai, SQLAlchemy ko 'postgresql+psycopg://' chahiye."""
    url = (url or "").strip()
    if url.startswith("postgres://"):
        url = "postgresql://" + url[len("postgres://"):]
    if url.startswith("postgresql://"):
        url = "postgresql+psycopg://" + url[len("postgresql://"):]
    return url


def engine_kwargs(url: str) -> dict:
    if url.startswith("sqlite"):
        return {"connect_args": {"check_same_thread": False}}
    return {"pool_pre_ping": True}  # toote hue connections khud theek ho jate hain


DATABASE_URL = normalize_database_url(settings.database_url)

engine = create_engine(DATABASE_URL, **engine_kwargs(DATABASE_URL))
SessionLocal = sessionmaker(bind=engine, autoflush=False, autocommit=False)

import os

from app.core.config import settings

DEFAULT_URL = "http://127.0.0.1:8000"
_transport = None  # tests yahan fake transport lagate hain


class ApiError(Exception):
    """API se rabta ya jawab me masla (dashboard ye message seedha dikhata hai)."""


def base_url() -> str:
    return os.getenv("API_URL", DEFAULT_URL).rstrip("/")


def _detail(response: httpx.Response) -> str:
    try:
        detail = response.json().get("detail")
    except (ValueError, AttributeError):
        detail = None
    if isinstance(detail, str) and detail:
        return detail
    return f"API error {response.status_code}"


def _forward_headers() -> dict:
    """Dashboard ke visitor ka IP X-Forwarded-For me forward karti hai (rate limit sahi chale)."""
    try:
        import ipaddress

        import streamlit as st

        raw = st.context.headers.get("x-forwarded-for", "")
        first = raw.split(",")[0].strip()
        ipaddress.ip_address(first)  # galat/ganda value ho to ValueError, hum chhor dete hain
    except Exception:
        return {}
    return {"X-Forwarded-For": first}


def _request(
    method: str, path: str, timeout: float = 20.0, extra_headers: dict | None = None, **kwargs
):
    try:
        headers = {"X-API-Key": settings.api_key} if settings.api_key else {}
        headers.update(_forward_headers())
        headers.update(extra_headers or {})
        with httpx.Client(
            base_url=base_url(), timeout=timeout, transport=_transport, headers=headers
        ) as client:
            response = client.request(method, path, **kwargs)
    except httpx.ConnectError:
        raise ApiError(
            "API server se connection nahi ho paya. Pehle server chalayein: "
            "uvicorn app.main:app --reload"
        ) from None
    except httpx.TimeoutException:
        raise ApiError("API ne waqt par jawab nahi diya (timeout). Dobara try karein.") from None
    except httpx.HTTPError as e:
        raise ApiError(f"Network error: {type(e).__name__}") from None

    if response.status_code >= 400:
        raise ApiError(_detail(response))
    return response.json()

def _load_bookmarks() -> None:
    """Keep the logged-in user's saved story ids in session state (reloaded per user)."""
    token = st.session_state.get("token")
    if not token:
        return
    user = st.session_state.get("user_email")
    if "bookmark_ids" in st.session_state and st.session_state.get("bookmark_user") == user:
        return
    try:
        st.session_state["bookmark_ids"] = set(api_client.my_bookmarks(token)["story_ids"])
    except api_client.ApiError:
        st.session_state["bookmark_ids"] = set()
    st.session_state["bookmark_user"] = user

"""Reusable ORACLE UI pieces. The *_html helpers are pure so they can be tested offline."""
from html import escape

import altair as alt
import pandas as pd
import streamlit as st

from dashboard import api_client, utils
from dashboard.oracle import theme


def badge(text, color: str) -> str:
    return f'<span class="badge" style="background:{color}">{escape(str(text))}</span>'


def kpi_html(label, value, accent: str, sub: str = "") -> str:
    sub_html = f'<div class="sub">{escape(str(sub))}</div>' if sub else ""
    return (
        f'<div class="kpi" style="--accent:{accent}">'
        f'<div class="label">{escape(str(label))}</div>'
        f'<div class="value">{escape(str(value))}</div>{sub_html}</div>'
    )


def kpi_grid(items) -> str:
    return '<div class="kpi-grid">' + "".join(kpi_html(*i) for i in items) + "</div>"


def status_html(label: str, value: str, state: str) -> str:
    color = theme.STATE_COLORS.get(state, theme.STATE_COLORS["neutral"])
    return (
        '<div class="status">'
        f'<div class="s-label">{escape(label)}</div>'
        f'<div class="s-value"><span class="dot" style="--c:{color}"></span>{escape(value)}</div>'
        "</div>"
    )


def status_grid_html(checks: list[dict]) -> str:
    cells = "".join(status_html(c["label"], c["value"], c["state"]) for c in checks)
    return f'<div class="status-grid">{cells}</div>'


def empty_html(title: str, hint: str = "") -> str:
    return f'<div class="empty"><b>{escape(title)}</b>{escape(hint)}</div>'


def claim_flag(story: dict) -> bool:
    return any(c.get("status") == "disputed" for c in story.get("claims") or [])


def story_badges_html(story: dict) -> str:
    medium = theme.PRIORITY_COLORS["Medium"]
    parts = [
        badge(story.get("priority", "Medium"),
              theme.PRIORITY_COLORS.get(story.get("priority"), medium)),
        badge(story.get("sentiment", "neutral"),
              theme.SENTIMENT_COLORS.get(story.get("sentiment"), theme.SENTIMENT_COLORS["neutral"])),
        badge(story.get("credibility", "Single source"), "#334155"),
    ]
    if story.get("category"):
        parts.append(badge(story["category"], "#0E7490"))
    if claim_flag(story):
        parts.append(badge("Details differ", theme.PRIORITY_COLORS["High"]))
    for k in story.get("matched_interests") or []:
        parts.append(badge("match: " + k, "#6D28D9"))
    return " ".join(parts)


def story_row_html(story: dict) -> str:
    color = theme.PRIORITY_COLORS.get(story.get("priority"), theme.PRIORITY_COLORS["Medium"])
    return (
        '<div class="row-item">'
        f'{badge(story.get("priority", "Medium"), color)}'
        f'<span class="row-title">{utils.html_safe(story.get("title", ""))}</span>'
        f'<span class="row-time">{escape(utils.format_time(story.get("created_at")))}</span>'
        "</div>"
    )


def page_header(title: str, subtitle: str = "") -> None:
    st.markdown(
        f'<div class="page-title">{escape(title)}</div>'
        f'<div class="page-sub">{escape(subtitle)}</div>',
        unsafe_allow_html=True,
    )


def empty_state(title: str, hint: str = "") -> None:
    st.markdown(empty_html(title, hint), unsafe_allow_html=True)


def _thumbnail(story: dict) -> None:
    url = str(story.get("image_url") or "")
    if url.startswith(("http://", "https://")):
        st.image(url, width=280)


def _bookmark_button(story: dict, key: str, index: int) -> None:
    token = st.session_state.get("token")
    sid = story.get("id")
    if not token or sid is None:
        return
    saved = sid in st.session_state.get("bookmark_ids", set())
    if st.button("★ Saved" if saved else "☆ Save", key=f"bm_{key}_{index}"):
        try:
            if saved:
                api_client.remove_bookmark(token, sid)
            else:
                api_client.add_bookmark(token, sid)
        except api_client.ApiError as e:
            st.warning(str(e))
            return
        ids = set(st.session_state.get("bookmark_ids", set()))
        ids.symmetric_difference_update({sid})
        st.session_state["bookmark_ids"] = ids
        st.rerun()


def story_card(story: dict, index: int, key: str) -> None:
    with st.container(key=f"card_{key}_{index}"):
        _thumbnail(story)
        st.markdown(story_badges_html(story), unsafe_allow_html=True)
        st.markdown(
            f'<div class="story-title">{index}. {utils.html_safe(story.get("title", ""))}</div>',
            unsafe_allow_html=True,
        )
        st.markdown(utils.md_safe(story.get("summary", "")))
        if story.get("claims"):
            st.markdown(utils.claims_markdown(story["claims"]))
        sources = story.get("sources") or []
        st.caption(
            f"{utils.format_time(story.get('created_at'))} | Sources: "
            + utils.sources_markdown(sources)
        )
        link = next(
            (s["link"] for s in sources if str(s.get("link", "")).startswith(("http://", "https://"))),
            None,
        )
        if link:
            st.link_button("Read full story", link, key=f"read_{key}_{index}")
        _bookmark_button(story, key, index)


def render_sources(sources) -> None:
    if not sources:
        return
    with st.expander("Sources"):
        for s in sources:
            links = utils.sources_markdown(s.get("sources", []))
            st.markdown(f"**[{s['n']}]** {utils.md_safe(s['title'])}  \n{links}")


def bar_chart(rows, category: str, value: str, colors: dict | None = None,
              height: int = 260, horizontal: bool = False):
    df = pd.DataFrame(rows)
    if colors:
        color = alt.Color(
            f"{category}:N",
            scale=alt.Scale(domain=list(colors), range=list(colors.values())),
            legend=None,
        )
    else:
        color = alt.value(theme.PRIMARY)
    if horizontal:
        x, y = alt.X(f"{value}:Q", title=None), alt.Y(f"{category}:N", sort="-x", title=None)
    else:
        x, y = alt.X(f"{category}:N", sort=None, title=None), alt.Y(f"{value}:Q", title=None)
    return (
        alt.Chart(df)
        .mark_bar(cornerRadiusEnd=4)
        .encode(x=x, y=y, color=color, tooltip=[category, value])
        .properties(height=height)
        .configure(background="transparent")
        .configure_axis(labelColor=theme.MUTED, titleColor=theme.MUTED,
                        gridColor=theme.BORDER, domainColor=theme.BORDER)
        .configure_view(strokeWidth=0)
    )

class Credentials(BaseModel):
    email: str = Field(min_length=5, max_length=254)
    password: str = Field(min_length=8, max_length=128)


class UserInterests(BaseModel):
    interests: str = Field(default="", max_length=1000)


def _secret() -> str:
    if not settings.auth_secret:
        raise HTTPException(status_code=503, detail="Accounts band hain: AUTH_SECRET set nahi hai")
    return settings.auth_secret


def _throttle(request: Request, email: str | None = None) -> None:
    """Bruteforce rokne ke liye: IP aur (login par) email, dono par limit."""
    limit, window = settings.auth_rate_limit, settings.auth_rate_window_seconds
    keys = [f"auth-ip:{client_ip(request, settings.trust_proxy_headers)}"]
    if email:
        keys.append(f"auth-email:{email[:254]}")
    for key in keys:
        wait = ratelimit.hit(key, limit, window)
        if wait:
            raise HTTPException(
                status_code=429,
                detail="Bohat zyada koshishein. Thori der baad dobara try karein",
                headers={"Retry-After": str(int(wait) + 1)},
            )


def _issue(user: dict) -> dict:
    token = create_token(user["id"], _secret(), settings.auth_token_hours * 3600)
    return {"access_token": token, "token_type": "bearer", "user": user}


def get_current_user(authorization: str | None = Header(default=None)) -> dict:
    secret = _secret()
    scheme, _, token = (authorization or "").partition(" ")
    user_id = verify_token(token.strip(), secret) if scheme.lower() == "bearer" else None
    user = users.get_user(user_id) if user_id is not None else None
    if user is None:
        raise HTTPException(status_code=401, detail="Login chahiye ya token expire ho gaya")
    return user


@router.post("/auth/register", status_code=201)
def register(req: Credentials, request: Request):
    _secret()
    _throttle(request)
    email = users.normalize_email(req.email)
    if not users.is_valid_email(email):
        raise HTTPException(status_code=400, detail="Email sahi nahi hai")
    try:
        user = users.create_user(email, req.password)
    except users.UserExistsError:
        raise HTTPException(status_code=409, detail="Ye email pehle se registered hai")
    return _issue(user)


@router.post("/auth/login")
def login(req: Credentials, request: Request):
    _secret()
    _throttle(request, users.normalize_email(req.email))
    user = users.authenticate(req.email, req.password)
    if user is None:
        raise HTTPException(status_code=401, detail="Email ya password ghalat hai")
    return _issue(user)

        raise HTTPException(status_code=404, detail="User nahi mila")

"""Read-only demo mode: DEMO_MODE=true par data badalne wale requests band.

Sirf GET aur do read-only POST (/chat, /research) chalte hain.
"""
from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse

from app.core.config import settings

SAFE_METHODS = {"GET", "HEAD", "OPTIONS"}
ALLOWED_POSTS = {"/chat", "/research", "/research/web"}


def is_blocked(method: str, path: str) -> bool:
    if method.upper() in SAFE_METHODS:
        return False
    return not (method.upper() == "POST" and path.rstrip("/") in ALLOWED_POSTS)


def add_demo_guard(app: FastAPI) -> None:
    @app.middleware("http")
    async def demo_guard(request: Request, call_next):
        if settings.demo_mode and is_blocked(request.method, request.url.path):
            return JSONResponse(
                {"detail": "Demo mode: ye instance sirf padhne ke liye hai"}, status_code=403
            )
        return await call_next(request)

"""Request id, access log aur basic security headers."""
import logging
import re
import time
import uuid

from fastapi import FastAPI, Request
from fastapi.responses import JSONResponse

from app.core import error_alerts
from app.core.logging_setup import request_id_var

log = logging.getLogger("app.http")
_ID_RE = re.compile(r"^[A-Za-z0-9._-]{8,64}$")
SECURITY_HEADERS = {
    "X-Content-Type-Options": "nosniff",
    "X-Frame-Options": "DENY",
    "Referrer-Policy": "no-referrer",
}


def add_http_middleware(app: FastAPI) -> None:
    @app.middleware("http")
    async def request_wrapper(request: Request, call_next):
        incoming = request.headers.get("x-request-id", "")
        request_id = incoming if _ID_RE.match(incoming) else uuid.uuid4().hex
        token = request_id_var.set(request_id)
        start = time.perf_counter()
        fields = {"method": request.method, "path": request.url.path}
        try:
            try:
                response = await call_next(request)
            except Exception as exc:
                log.exception("unhandled error", extra={"fields": fields})
                error_alerts.report(
                    f"Server error 500: {request.method} {request.url.path}",
                    f"request_id={request_id} error={type(exc).__name__}",
                )
                response = JSONResponse(
                    {"detail": "Server me masla aaya", "request_id": request_id}, status_code=500
                )
            fields["status"] = response.status_code
            fields["duration_ms"] = round((time.perf_counter() - start) * 1000, 1)
            level = logging.DEBUG if request.url.path == "/health" else logging.INFO
            log.log(level, "request", extra={"fields": fields})
            response.headers["X-Request-ID"] = request_id
            for name, value in SECURITY_HEADERS.items():
                response.headers.setdefault(name, value)
            return response
        finally:
            request_id_var.reset(token)

import hmac

from fastapi import Header, HTTPException

from app.core.config import settings


def require_api_key(x_api_key: str | None = Header(default=None)) -> None:
    """API_KEY set ho to har request me X-API-Key header chahiye. Khali ho to koi rok nahi."""
    expected = settings.api_key
    if not expected:
        return
    if not x_api_key or not hmac.compare_digest(x_api_key.encode(), expected.encode()):
        raise HTTPException(status_code=401, detail="API key missing ya ghalat hai")

"""Web-source check endpoint (Gemini Google Search grounding)."""
from fastapi import APIRouter, HTTPException, Request
from pydantic import BaseModel, Field

from app.agents.research import ResearchError
from app.agents.web_check import web_check
from app.core import ratelimit
from app.core.config import settings
from app.core.ratelimit import client_ip

router = APIRouter()


class WebCheckRequest(BaseModel):
    text: str = Field(min_length=3, max_length=500)


@router.post("/research/web")
def research_web(req: WebCheckRequest, request: Request):
    """Claim ya topic ko web sources se check karta hai (links ke saath)."""
    text = req.text.strip()
    if len(text) < 3:
        raise HTTPException(status_code=400, detail="Text bohat chhota hai")
    wait = ratelimit.hit(f"webcheck:{client_ip(request, settings.trust_proxy_headers)}", 20, 3600)
    if wait:
        raise HTTPException(
            status_code=429,
            detail="Bohat zyada koshishein. Thori der baad dobara try karein",
            headers={"Retry-After": str(int(wait) + 1)},
        )
    try:
        return web_check(text)
    except ValueError as e:  # GEMINI_API_KEY set nahi
        raise HTTPException(status_code=503, detail=str(e))
    except ResearchError as e:
        raise HTTPException(status_code=502, detail=str(e))
